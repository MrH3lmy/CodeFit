class Exercise {
    private final int limit;
    private int active;

    Exercise(int n) {
        limit = n;
    }

    public boolean enter() {
        if (active > limit) return false;
        active++;
        return true;
    }

    public void leave() {
        active--;
    }

    public int used() {
        return active;
    }
}

public class Main {
    static void check(boolean value) {
        if (!value) throw new AssertionError("Contract violated");
    }

    static void parallel(Runnable a, Runnable b) throws Exception {
        var failure = new java.util.concurrent.atomic.AtomicReference<Throwable>();
        var gate = new java.util.concurrent.CountDownLatch(1);
        Thread left =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                a.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        Thread right =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                b.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        left.setDaemon(true);
        right.setDaemon(true);
        left.start();
        right.start();
        gate.countDown();
        left.join(3000);
        right.join(3000);
        if (left.isAlive() || right.isAlive()) throw new AssertionError("Observed nontermination");
        if (failure.get() != null) throw new AssertionError("Worker failed", failure.get());
    }

    public static void main(String[] args) throws Exception {
        String mode =
                new String(System.in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8)
                        .strip();
        if (mode.equals("concurrency")) {
            Exercise e = new Exercise(1);
            java.util.concurrent.atomic.AtomicInteger wins =
                    new java.util.concurrent.atomic.AtomicInteger();
            parallel(
                    () -> {
                        if (e.enter()) wins.incrementAndGet();
                    },
                    () -> {
                        if (e.enter()) wins.incrementAndGet();
                    });
            check(wins.get() == 1);
            check(e.used() == 1);
        } else {
            Exercise e = new Exercise(1);
            check(e.enter());
            check(!e.enter());
            check(e.used() == 1);
            e.leave();
            check(e.used() == 0);
            try {
                e.leave();
                throw new AssertionError();
            } catch (IllegalStateException expected) {
            }
        }
        System.out.print("OK\n");
    }
}
