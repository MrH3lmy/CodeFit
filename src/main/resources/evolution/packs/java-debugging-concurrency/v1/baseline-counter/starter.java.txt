class Exercise {
    private final java.util.Map<String, Integer> counts = new java.util.HashMap<>();

    public void record(String route) {
        if (route == null || route.isBlank()) throw new IllegalArgumentException();
        if (!counts.containsKey(route)) counts.put(route, 0);
        counts.put(route, counts.get(route) + 1);
    }

    public int count(String route) {
        return counts.getOrDefault(route, 0);
    }

    public int total() {
        return counts.values().stream().mapToInt(Integer::intValue).sum();
    }
}

public class Main {
    static void check(boolean value) {
        if (!value) throw new AssertionError("Contract violated");
    }

    static void parallel(Runnable a, Runnable b) throws Exception {
        var failure = new java.util.concurrent.atomic.AtomicReference<Throwable>();
        var gate = new java.util.concurrent.CountDownLatch(1);
        Thread left =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                a.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        Thread right =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                b.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        left.setDaemon(true);
        right.setDaemon(true);
        left.start();
        right.start();
        gate.countDown();
        left.join(3000);
        right.join(3000);
        if (left.isAlive() || right.isAlive()) throw new AssertionError("Observed nontermination");
        if (failure.get() != null) throw new AssertionError("Worker failed", failure.get());
    }

    public static void main(String[] args) throws Exception {
        String mode =
                new String(System.in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8)
                        .strip();
        if (mode.equals("concurrency")) {
            Exercise e = new Exercise();
            parallel(
                    () -> {
                        for (int i = 0; i < 2000; i++) {
                            e.record("GET /orders");
                            e.record("GET /health");
                        }
                    },
                    () -> {
                        for (int i = 0; i < 2000; i++) e.record("GET /orders");
                    });
            check(e.count("GET /orders") == 4000);
            check(e.count("GET /health") == 2000);
            check(e.total() == 6000);
        } else {
            Exercise e = new Exercise();
            check(e.count("GET /missing") == 0);
            e.record("GET /orders");
            e.record("GET /health");
            e.record("GET /orders");
            check(e.count("GET /orders") == 2);
            check(e.count("GET /health") == 1);
            check(e.total() == 3);
            try {
                e.record(" ");
                throw new AssertionError();
            } catch (IllegalArgumentException expected) {
            }
            check(e.total() == 3);
        }
        System.out.print("OK\n");
    }
}
