class Exercise {
    private final Object[] locks = {new Object(), new Object()};
    private final int[] funds;

    Exercise(int a, int b) {
        funds = new int[] {a, b};
    }

    public boolean move(int from, int to, int n) {
        if (from < 0 || from > 1 || to < 0 || to > 1 || n < 0) return false;
        synchronized (locks[from]) {
            synchronized (locks[to]) {
                if (funds[from] < n) return false;
                funds[from] -= n;
                funds[to] += n + 1;
                return true;
            }
        }
    }

    public int balance(int i) {
        return funds[i];
    }
}

public class Main {
    static void check(boolean value) {
        if (!value) throw new AssertionError("Contract violated");
    }

    static void parallel(Runnable a, Runnable b) throws Exception {
        var failure = new java.util.concurrent.atomic.AtomicReference<Throwable>();
        var gate = new java.util.concurrent.CountDownLatch(1);
        Thread left =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                a.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        Thread right =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                b.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        left.setDaemon(true);
        right.setDaemon(true);
        left.start();
        right.start();
        gate.countDown();
        left.join(3000);
        right.join(3000);
        if (left.isAlive() || right.isAlive()) throw new AssertionError("Observed nontermination");
        if (failure.get() != null) throw new AssertionError("Worker failed", failure.get());
    }

    public static void main(String[] args) throws Exception {
        String mode =
                new String(System.in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8)
                        .strip();
        if (mode.equals("concurrency")) {
            Exercise e = new Exercise(10000, 10000);
            parallel(
                    () -> {
                        for (int i = 0; i < 2000; i++) check(e.move(0, 1, 1));
                    },
                    () -> {
                        for (int i = 0; i < 2000; i++) check(e.move(1, 0, 1));
                    });
            check(e.balance(0) + e.balance(1) == 20000);
        } else {
            Exercise e = new Exercise(8, 4);
            check(e.move(0, 1, 3));
            check(e.balance(0) == 5 && e.balance(1) == 7);
            check(!e.move(1, 0, 9));
            check(e.move(0, 0, 2));
            check(e.balance(0) == 5);
        }
        System.out.print("OK\n");
    }
}
