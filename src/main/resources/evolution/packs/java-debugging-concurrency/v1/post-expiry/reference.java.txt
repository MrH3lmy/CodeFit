class Exercise {
    private final java.util.Map<String, Long> live = new java.util.HashMap<>();
    private final java.util.Set<String> used = new java.util.HashSet<>();

    public synchronized boolean acquire(String id, long deadline, long now) {
        if (now < 0 || deadline <= now || !used.add(id)) return false;
        live.put(id, deadline);
        return true;
    }

    public synchronized int expire(long now) {
        int before = live.size();
        live.values().removeIf(deadline -> deadline <= now);
        return before - live.size();
    }

    public synchronized boolean release(String id) {
        return live.remove(id) != null;
    }

    public synchronized int active() {
        return live.size();
    }
}
