class Exercise {
    private final java.util.Map<String, Long> live = new java.util.HashMap<>();

    public boolean acquire(String id, long deadline, long now) {
        if (now < 0 || deadline <= now || live.containsKey(id)) return false;
        live.put(id, deadline);
        return true;
    }

    public int expire(long now) {
        int before = live.size();
        live.values().removeIf(deadline -> deadline < now);
        return before - live.size();
    }

    public boolean release(String id) {
        return live.remove(id) != null;
    }

    public int active() {
        return live.size();
    }
}

public class Main {
    static void check(boolean value) {
        if (!value) throw new AssertionError("Contract violated");
    }

    static void parallel(Runnable a, Runnable b) throws Exception {
        var failure = new java.util.concurrent.atomic.AtomicReference<Throwable>();
        var gate = new java.util.concurrent.CountDownLatch(1);
        Thread left =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                a.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        Thread right =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                b.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        left.setDaemon(true);
        right.setDaemon(true);
        left.start();
        right.start();
        gate.countDown();
        left.join(3000);
        right.join(3000);
        if (left.isAlive() || right.isAlive()) throw new AssertionError("Observed nontermination");
        if (failure.get() != null) throw new AssertionError("Worker failed", failure.get());
    }

    public static void main(String[] args) throws Exception {
        String mode =
                new String(System.in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8)
                        .strip();
        if (mode.equals("concurrency")) {
            Exercise e = new Exercise();
            e.acquire("a", 5, 0);
            java.util.concurrent.atomic.AtomicInteger removed =
                    new java.util.concurrent.atomic.AtomicInteger();
            parallel(() -> removed.addAndGet(e.expire(5)), () -> removed.addAndGet(e.expire(5)));
            check(removed.get() == 1 && e.active() == 0);
        } else {
            Exercise e = new Exercise();
            check(e.acquire("a", 5, 0));
            check(e.expire(5) == 1);
            check(e.expire(5) == 0);
            check(!e.release("a"));
            check(!e.acquire("a", 9, 5));
            check(e.active() == 0);
        }
        System.out.print("OK\n");
    }
}
