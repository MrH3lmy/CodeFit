class Exercise {
    private final java.util.ArrayDeque<String> q = new java.util.ArrayDeque<>();
    private final java.util.Set<String> flight = new java.util.HashSet<>();
    private int complete;

    public boolean enqueue(String id) {
        q.add(id);
        return true;
    }

    public String claim() {
        String id = q.poll();
        if (id != null) flight.add(id);
        return id;
    }

    public boolean ack(String id) {
        flight.remove(id);
        complete++;
        return true;
    }

    public int pending() {
        return q.size();
    }

    public int inFlight() {
        return flight.size();
    }

    public int done() {
        return complete;
    }
}

public class Main {
    static void check(boolean value) {
        if (!value) throw new AssertionError("Contract violated");
    }

    static void parallel(Runnable a, Runnable b) throws Exception {
        var failure = new java.util.concurrent.atomic.AtomicReference<Throwable>();
        var gate = new java.util.concurrent.CountDownLatch(1);
        Thread left =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                a.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        Thread right =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                b.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        left.setDaemon(true);
        right.setDaemon(true);
        left.start();
        right.start();
        gate.countDown();
        left.join(3000);
        right.join(3000);
        if (left.isAlive() || right.isAlive()) throw new AssertionError("Observed nontermination");
        if (failure.get() != null) throw new AssertionError("Worker failed", failure.get());
    }

    public static void main(String[] args) throws Exception {
        String mode =
                new String(System.in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8)
                        .strip();
        if (mode.equals("concurrency")) {
            Exercise e = new Exercise();
            for (int i = 0; i < 100; i++) e.enqueue("j" + i);
            parallel(
                    () -> {
                        String id;
                        while ((id = e.claim()) != null) e.ack(id);
                    },
                    () -> {
                        String id;
                        while ((id = e.claim()) != null) e.ack(id);
                    });
            check(e.done() == 100 && e.pending() == 0 && e.inFlight() == 0);
        } else {
            Exercise e = new Exercise();
            check(e.enqueue("a"));
            check(!e.enqueue("a"));
            check(!e.ack("a"));
            check("a".equals(e.claim()));
            check(e.ack("a"));
            check(!e.ack("a"));
            check(!e.enqueue("a"));
            check(e.pending() == 0 && e.inFlight() == 0 && e.done() == 1);
        }
        System.out.print("OK\n");
    }
}
