class Exercise {
    public static int[] page(int[] ids, int after, int limit) {
        if (limit <= 0) return new int[0];
        return java.util.Arrays.stream(ids)
                .skip(Math.max(0, (long) after + 1))
                .limit(limit)
                .toArray();
    }
}

public class Main {
    static void check(boolean value) {
        if (!value) throw new AssertionError("Contract violated");
    }

    static void parallel(Runnable a, Runnable b) throws Exception {
        var failure = new java.util.concurrent.atomic.AtomicReference<Throwable>();
        var gate = new java.util.concurrent.CountDownLatch(1);
        Thread left =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                a.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        Thread right =
                new Thread(
                        () -> {
                            try {
                                gate.await();
                                b.run();
                            } catch (Throwable t) {
                                failure.set(t);
                            }
                        });
        left.setDaemon(true);
        right.setDaemon(true);
        left.start();
        right.start();
        gate.countDown();
        left.join(3000);
        right.join(3000);
        if (left.isAlive() || right.isAlive()) throw new AssertionError("Observed nontermination");
        if (failure.get() != null) throw new AssertionError("Worker failed", failure.get());
    }

    public static void main(String[] args) throws Exception {
        String mode =
                new String(System.in.readAllBytes(), java.nio.charset.StandardCharsets.UTF_8)
                        .strip();
        if (mode.equals("concurrency")) {
            check(
                    java.util.Arrays.equals(
                            Exercise.page(new int[] {2, 8, 19}, 8, 2), new int[] {19}));
            check(
                    java.util.Arrays.equals(
                            Exercise.page(new int[] {2, 8, 19}, 3, 1), new int[] {8}));
            check(Exercise.page(new int[] {2}, 0, 0).length == 0);
        } else {
            check(
                    java.util.Arrays.equals(
                            Exercise.page(new int[] {2, 8, 19}, 8, 2), new int[] {19}));
            check(
                    java.util.Arrays.equals(
                            Exercise.page(new int[] {2, 8, 19}, 3, 1), new int[] {8}));
            check(Exercise.page(new int[] {2}, 0, 0).length == 0);
        }
        System.out.print("OK\n");
    }
}
